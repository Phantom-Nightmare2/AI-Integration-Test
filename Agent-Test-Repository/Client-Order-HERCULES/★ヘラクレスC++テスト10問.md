    int total = 0;

        ++count;
        ++total;

        return {count, total};

    std::atomic<bool> inconsistent{false};

            auto snapshot = stats.snapshot();

                inconsistent.store(true);

            for (int i = 0; i < 10000; ++i) {
                stats.add();
            }
        });
    }
    for (auto& t : writers) t.join();
    done.store(true);
    observer.join();
    auto result = stats.snapshot();

              << " Total=" << result.second

## Q07：condition_variableと安全なキュー終了

**難易度：高難度**

### クライアント要望

3つのワーカースレッドが共有キューから仕事を取り出します。待機からの誤った復帰、終了通知の不足、終了後の投入受付が問題です。キューの仕事をすべて処理してから全スレッドを停止させてください。

### 元コード（編集禁止）

```cpp
#include <atomic>
#include <condition_variable>
#include <iostream>
#include <mutex>
#include <queue>
#include <thread>
#include <vector>

class WorkQueue {
    std::queue<int> jobs;
    std::mutex mu;
    std::condition_variable cv;
    bool closed = false;
public:
    void push(int value) {
        std::lock_guard<std::mutex> lock(mu);
        jobs.push(value);
        cv.notify_one();
    }
    bool pop(int& value) {
        std::unique_lock<std::mutex> lock(mu);
        if (jobs.empty()) cv.wait(lock);
        if (jobs.empty()) return false;
        value = jobs.front();
        jobs.pop();
        return true;
    }
    void close() {
        std::lock_guard<std::mutex> lock(mu);
        closed = true;
        cv.notify_one();
    }
};

int main() {
    WorkQueue queue;
    std::atomic<int> sum{0};
    std::atomic<int> processed{0};
    std::vector<std::thread> workers;
    for (int i = 0; i < 3; ++i) {
        workers.emplace_back([&] {
            int value;
            while (queue.pop(value)) {
                sum.fetch_add(value);
                processed.fetch_add(1);
            }
        });
    }
    for (int i = 1; i <= 100; ++i) queue.push(i);
    queue.close();
    for (auto& t : workers) t.join();
    std::cout << "Processed=" << processed
              << " Sum=" << sum << '\n';
}
```

### 期待結果・合格条件

Processed=100 Sum=5050。全ワーカーが終了し、ハングしない。close後の新しいpushは拒否される。

### 必須要件

- 条件変数の待機条件を正しく扱う
- 処理待ちのキューを空にしてから全ワーカーを終了する
- closeを複数回呼んでも破綻しない
- 終了後のpush拒否、空キュー終了、複数回実行を検証する
- ハングの可能性がある元コードは外部タイムアウト付きで実行する

## Q10：スレッドプールの停止・例外伝播・投入拒否

**難易度：最難関**

### クライアント要望

実案件向けの小型スレッドプールです。停止処理のハング、実行待ちタスクの消失、停止後の投入許可、待機条件の誤りを修正してください。公開APIを保ち、タスクが例外を投げても結果をfuture経由で回収できるようにしてください。

### 元コード（編集禁止）

```cpp
#include <condition_variable>
#include <functional>
#include <future>
#include <iostream>
#include <memory>
#include <mutex>
#include <queue>
#include <stdexcept>
#include <thread>
#include <type_traits>
#include <utility>
#include <vector>

class ThreadPool {
    std::vector<std::thread> workers_;
    std::queue<std::function<void()>> jobs_;
    std::mutex mu_;
    std::condition_variable cv_;
    bool stopped_ = false;

    void worker() {
        while (true) {
            std::unique_lock<std::mutex> lock(mu_);
            cv_.wait(lock);
            if (stopped_) return;
            if (jobs_.empty()) continue;
            auto job = std::move(jobs_.front());
            jobs_.pop();
            lock.unlock();
            job();
        }
    }

public:
    explicit ThreadPool(int count) {
        for (int i = 0; i < count; ++i) {
            workers_.emplace_back([this] { worker(); });
        }
    }

    template<class F>
    auto submit(F task) -> std::future<std::invoke_result_t<F>> {
        using R = std::invoke_result_t<F>;
        auto packaged =
            std::make_shared<std::packaged_task<R()>>(std::move(task));
        auto future = packaged->get_future();
        {
            std::lock_guard<std::mutex> lock(mu_);
            jobs_.push([packaged] { (*packaged)(); });
        }
        cv_.notify_one();
        return future;
    }

    void stop() {
        {
            std::lock_guard<std::mutex> lock(mu_);
            stopped_ = true;
        }
        cv_.notify_one();
        for (auto& t : workers_) {
            if (t.joinable()) t.join();
        }
    }

    ~ThreadPool() { stop(); }
};

int main() {
    ThreadPool pool(3);
    std::vector<std::future<int>> results;
    for (int i = 1; i <= 20; ++i) {
        results.push_back(pool.submit([i] {
            if (i == 13) throw std::runtime_error("job 13 failed");
            return i * i;
        }));
    }
    pool.stop();
    int succeeded = 0, failed = 0, sum = 0;
    for (auto& f : results) {
        try {
            sum += f.get();
            ++succeeded;
        } catch (const std::exception&) {
            ++failed;
        }
    }
    std::cout << "Success=" << succeeded
              << " Failed=" << failed
              << " Sum=" << sum << '\n';
    try {
        auto future = pool.submit([] { return 1; });
        std::cout << "Rejected=no\n";
    } catch (const std::exception&) {
        std::cout << "Rejected=yes\n";
    }
}
```

Success=19 Failed=1 Sum=2701、および Rejected=yes。保留中タスクを取りこぼさず終了し、スレッドがハングしない。

### 必須要件

- 条件変数の待機条件を正しく定義する
- stop時は受理済みタスクを最後まで実行し全スレッドをjoinする
- 停止後のsubmitを即座に例外等で拒否する
- stopを複数回呼んでも安全である
- タスクの例外をfuture経由で呼び出し側に伝える
- 複数生産者から同時submitしてもタスクが失われないことを検証する
- ハングする元コードは外部タイムアウト付きで検証する
